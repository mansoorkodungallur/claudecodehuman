def simple_chat(user_message):
    """Send a single message and get a response."""
    try:
        response = client.chat.completions.create(
            model="gpt-4",  # or "gpt-3.5-turbo" for faster/cheaper
            messages=[
                {"role": "user", "content": user_message}
            ],
            temperature=0.7,  # 0-2, higher = more creative
            max_tokens=500    # Limit response length
        )

        return response.choices[0].message.content

    except Exception as e:
        return f"Error: {str(e)}"

# Example usage
result = simple_chat("Explain quantum computing in 3 sentences")
print(result)
3. Conversation with Context 
class ChatBot:
    """A stateful chatbot that remembers conversation context."""

    def __init__(self, system_prompt="You are a helpful assistant."):
        self.messages = [
            {"role": "system", "content": system_prompt}
        ]

    def chat(self, user_message):
        """Add user message, get response, and maintain history."""
        # Add user message to history
        self.messages.append({
            "role": "user",
            "content": user_message
        })

        try:
            # Get response
            response = client.chat.completions.create(
                model="gpt-4",
                messages=self.messages,
                temperature=0.7,
                max_tokens=500
            )

            # Extract assistant's reply
            assistant_message = response.choices[0].message.content

            # Add to history
            self.messages.append({
                "role": "assistant",
                "content": assistant_message
            })

            return assistant_message

        except Exception as e:
            return f"Error: {str(e)}"

    def reset(self, system_prompt=None):
        """Clear conversation history."""
        if system_prompt:
            self.messages = [{"role": "system", "content": system_prompt}]
        else:
            self.messages = [self.messages[0]]  # Keep system message

# Example usage
bot = ChatBot("You are a Python expert who explains concepts clearly.")
print(bot.chat("What are decorators?"))
print(bot.chat("Can you show me an example?"))  # Remembers context!
4. Error Handling & Retries 
import time
from openai import OpenAI, APIError, RateLimitError, APIConnectionError

def robust_chat(user_message, max_retries=3):
    """Chat with automatic retry logic for common errors."""

    for attempt in range(max_retries):
        try:
            response = client.chat.completions.create(
                model="gpt-4",
                messages=[{"role": "user", "content": user_message}],
                timeout=30  # 30 second timeout
            )
            return response.choices[0].message.content

        except RateLimitError:
            # Hit rate limit, wait and retry
            wait_time = (2 ** attempt)  # Exponential backoff
            print(f"Rate limited. Waiting {wait_time}s...")
            time.sleep(wait_time)

        except APIConnectionError:
            # Network issue
            print(f"Connection error. Retry {attempt + 1}/{max_retries}")
            time.sleep(2)

        except APIError as e:
            # General API error
            print(f"API Error: {e}")
            return None

        except Exception as e:
            # Unexpected error
            print(f"Unexpected error: {e}")
            return None

    return "Failed after max retries"

# Example usage
result = robust_chat("Write a haiku about programming")
print(result)
5. Streaming Responses 
def streaming_chat(user_message):
    """Stream the response token by token (like ChatGPT web interface)."""

    try:
        stream = client.chat.completions.create(
            model="gpt-4",
            messages=[{"role": "user", "content": user_message}],
            stream=True  # Enable streaming
        )

        full_response = ""

        for chunk in stream:
            # Get the content from this chunk
            if chunk.choices[0].delta.content is not None:
                content = chunk.choices[0].delta.content
                full_response += content
                print(content, end='', flush=True)  # Print as it arrives

        print()  # New line at end
        return full_response

    except Exception as e:
        print(f"Error: {e}")
        return None

# Example usage
streaming_chat("Write a short story about a robot learning to paint")

Token Management & Cost Optimization

Understanding tokens is crucial for cost control and performance optimization. Every API call costs money based on token usage.

What Are Tokens?

Tokens are chunks of text that the AI processes. Roughly speaking:

  • 1 token ≈ 4 characters in English
  • 1 token ≈ ¾ of a word on average
  • 100 tokens ≈ 75 words

Both input (prompt) and output (response) count toward your usage.

💰 Cost Optimization Strategies

1

Choose the Right Model

Use GPT-3.5 Turbo for simple tasks (10-20x cheaper). Reserve GPT-4 for complex reasoning.

Example: Summarization, simple Q&A → GPT-3.5
Complex analysis, creative writing → GPT-4
2

Limit Response Length

Use max_tokens to cap response size. Don't pay for more than you need.

Code: 
max_tokens=200  # For short answers
3

Compress Your Prompts

Remove unnecessary words while maintaining clarity. Every token counts.

Before: "Could you please write a summary..."
After: "Summarize:"
4

Cache Common Responses

For repeated queries, cache results locally instead of calling API every time.

Savings: If same question asked 100x, pay once instead of 100x.
5

Batch Process When Possible

Combine multiple simple requests into one API call where logical.

Instead of: 10 calls for 10 summaries
Do: 1 call: "Summarize each of these 10 texts..."
6

Set Usage Limits

Use OpenAI's usage limits to prevent unexpected bills from bugs or abuse.

Where: OpenAI Dashboard → Usage Limits
⚠️ Real-World Costs: A typical chatbot with 1000 daily users averaging 10 messages each (500 tokens per exchange) on GPT-4 would cost ~$600/day. On GPT-3.5 Turbo, the same usage costs ~$30/day. Choose wisely!

Chaining & Parallelization

Build sophisticated workflows by combining multiple AI calls. This is how professional AI tools are built.

Common Workflow Patterns

🔗 Sequential Chain

Output of one call becomes input to the next. Use for multi-step transformations.

Raw Text
→
Summarize
→
Extract Key Points
→
Generate Action Items
Python Example: 
def sequential_workflow(text):
    # Step 1: Summarize
    summary = simple_chat(f"Summarize this in 2 paragraphs:\n{text}")

    # Step 2: Extract key points
    key_points = simple_chat(f"Extract 5 key points from:\n{summary}")

    # Step 3: Generate action items
    actions = simple_chat(f"Convert to action items:\n{key_points}")

    return {
        'summary': summary,
        'key_points': key_points,
        'actions': actions
    }

⚡ Parallel Processing

Run multiple independent calls simultaneously to save time.

Input Text
→
Summarize
→
Sentiment
→
Keywords
→
Combine Results
Python Example: 
import asyncio
from openai import AsyncOpenAI

async_client = AsyncOpenAI()

async def parallel_workflow(text):
    # Define all tasks
    tasks = [
        async_client.chat.completions.create(
            model="gpt-4",
            messages=[{"role": "user", "content": f"Summarize:\n{text}"}]
        ),
        async_client.chat.completions.create(
            model="gpt-4",
            messages=[{"role": "user", "content": f"Sentiment:\n{text}"}]
        ),
        async_client.chat.completions.create(
            model="gpt-4",
            messages=[{"role": "user", "content": f"Keywords:\n{text}"}]
        )
    ]

    # Run all in parallel
    results = await asyncio.gather(*tasks)

    return {
        'summary': results[0].choices[0].message.content,
        'sentiment': results[1].choices[0].message.content,
        'keywords': results[2].choices[0].message.content
    }

# Usage
asyncio.run(parallel_workflow("Your text here..."))

Custom GPT Creation

Custom GPTs allow you to create specialized AI assistants without coding. Perfect for recurring tasks with specific requirements.

What Are Custom GPTs?

Custom GPTs are personalized versions of ChatGPT that you can create for specific purposes. They have:

  • Custom instructions: Define behavior and personality
  • Knowledge files: Upload documents for reference
  • Capabilities: Enable/disable web browsing, image generation, code execution
  • Actions: Connect to external APIs

📋 Step-by-Step Creation Guide

Step 1

Define Your Use Case

Be crystal clear about what problem your GPT solves.

Examples:
  • Content Editor: Reviews blog posts for grammar, SEO, readability
  • Code Reviewer: Analyzes code for bugs, security, best practices
  • Recipe Generator: Creates recipes based on available ingredients
  • Interview Prep: Generates questions based on job description
Step 2

Write Instructions

The "Instructions" field is like a permanent system prompt. Be specific!

Template: 
You are a [role] specialized in [domain].

Your purpose is to [main goal].

When users provide input, you should:
1. [First action]
2. [Second action]
3. [Third action]

Format your responses as:
- [Format requirement 1]
- [Format requirement 2]

Constraints:
- [Constraint 1]
- [Constraint 2]

Tone: [Professional/Friendly/Technical/etc.]
Real Example (Content Editor GPT): 
You are an expert content editor with 10 years of experience in digital publishing.

Your purpose is to review and improve blog posts for clarity, engagement, and SEO.

When users provide a blog post, you should:
1. Identify grammar and spelling errors
2. Suggest improvements for readability (Flesch score 60+)
3. Optimize for SEO (keyword density, meta description)
4. Enhance engagement (hooks, CTAs, scanability)

Format your responses as:
- Issue Summary (bullet points)
- Detailed Feedback (section by section)
- Revised Version (full edited text)
- SEO Score (0-100 with explanation)

Constraints:
- Maintain the author's voice and style
- Keep changes minimal but impactful
- Explain why each change improves the content

Tone: Constructive and encouraging
Step 3

Upload Knowledge Files (Optional)

Provide reference documents your GPT can search and cite.

Good for:
  • Company style guides
  • Product documentation
  • Brand guidelines
  • Legal templates
  • Research papers
Supported formats: PDF, DOCX, TXT, code files

Here's the complete Python script for your content pipeline:

content_pipeline.py 
from openai import OpenAI
import json
import os
from datetime import datetime

client = OpenAI(api_key=os.environ.get("OPENAI_API_KEY"))

class ContentPipeline:
    def __init__(self, topic, audience, target_length=1500):
        self.topic = topic
        self.audience = audience
        self.target_length = target_length
        self.outline = None
        self.content = None
        self.seo_data = None

    def stage1_outline(self):
        """Generate comprehensive outline."""
        prompt = f"""You are an expert content strategist.

Create a detailed blog post outline for: "{self.topic}"

Requirements:
- Target audience: {self.audience}
- Article length: {self.target_length} words
- Include 5-7 main sections
- Each section should have 2-3 sub-points
- Suggest 3 attention-grabbing title options

Format as JSON:
{{
    "titles": ["option1", "option2", "option3"],
    "introduction": "hook text",
    "sections": [
        {{"title": "...", "sub_points": ["...", "..."]}}
    ],
    "conclusion": "summary",
    "meta_description": "...",
    "slug": "url-slug"
}}"""

        response = client.chat.completions.create(
            model="gpt-4",
            messages=[{"role": "user", "content": prompt}],
            temperature=0.7
        )

        self.outline = response.choices[0].message.content
        return self.outline

    def stage2_write(self):
        """Write full content based on outline."""
        if not self.outline:
            self.stage1_outline()

        prompt = f"""Write a complete blog post using this outline:

{self.outline}

Writing guidelines:
- Tone: Professional but conversational
- Style: Easy to scan with headers, lists, bold
- Include examples and data
- Use markdown formatting
- Length: {self.target_length} words

Write the complete post now."""

        response = client.chat.completions.create(
            model="gpt-4",
            messages=[{"role": "user", "content": prompt}],
            temperature=0.7,
            max_tokens=3000
        )

        self.content = response.choices[0].message.content
        return self.content

    def stage3_seo_optimize(self):
        """Analyze and optimize for SEO."""
        if not self.content:
            self.stage2_write()

        prompt = f"""Analyze this blog post for SEO:

{self.content}

Provide as JSON:
{{
    "seo_score": 0-100,
    "readability_score": 0-100,
    "keyword_density": "...",
    "issues": ["issue1", "issue2"],
    "meta_description": "155 chars",
    "title_tag": "60 chars",
    "improvements": ["suggestion1", "suggestion2"]
}}"""

        response = client.chat.completions.create(
            model="gpt-4",
            messages=[{"role": "user", "content": prompt}],
            temperature=0.3
        )

        self.seo_data = response.choices[0].message.content
        return self.seo_data

    def stage4_images(self):
        """Generate image suggestions."""
        prompt = f"""Based on this blog post, suggest 3 images:

{self.content}

For each provide JSON:
{{
    "images": [
        {{
            "placement": "section name",
            "purpose": "hero/illustration/diagram",
            "dalle_prompt": "detailed description for image generation",
            "alt_text": "SEO-optimized alt text"
        }}
    ]
}}"""

        response = client.chat.completions.create(
            model="gpt-4",
            messages=[{"role": "user", "content": prompt}],
            temperature=0.7
        )

        return response.choices[0].message.content

    def stage5_final_review(self):
        """Final editorial review."""
        prompt = f"""Perform final editorial review:

{self.content}

Check grammar, flow, accuracy. Provide:
- Issues found
- Publication readiness score (0-100)
- Any final tweaks needed

Keep response concise."""

        response = client.chat.completions.create(
            model="gpt-4",
            messages=[{"role": "user", "content": prompt}],
            temperature=0.3
        )

        return response.choices[0].message.content

    def run_full_pipeline(self, save_to_file=True):
        """Execute entire pipeline and save results."""
        print("🚀 Starting content pipeline...")

        print("\n📝 Stage 1: Generating outline...")
        self.stage1_outline()
        print("✅ Outline complete")

        print("\n✍️ Stage 2: Writing content...")
        self.stage2_write()
        print("✅ Content written")

        print("\n🔍 Stage 3: SEO optimization...")
        self.stage3_seo_optimize()
        print("✅ SEO analyzed")

        print("\n🖼️ Stage 4: Image suggestions...")
        images = self.stage4_images()
        print("✅ Images suggested")

        print("\n👀 Stage 5: Final review...")
        review = self.stage5_final_review()
        print("✅ Review complete")

        # Compile results
        results = {
            "topic": self.topic,
            "generated_at": datetime.now().isoformat(),
            "outline": self.outline,
            "content": self.content,
            "seo_data": self.seo_data,
            "images": images,
            "final_review": review
        }

        if save_to_file:
            filename = f"blog_{self.topic.replace(' ', '_').lower()}_{datetime.now().strftime('%Y%m%d')}.json"
            with open(filename, 'w') as f:
                json.dump(results, f, indent=2)
            print(f"\n💾 Saved to: {filename}")

        return results

# Usage Example
if __name__ == "__main__":
    pipeline = ContentPipeline(
        topic="The Future of AI in Healthcare",
        audience="Healthcare professionals and tech enthusiasts",
        target_length=2000
    )

    results = pipeline.run_full_pipeline()
    print("\n🎉 Pipeline complete! Check the generated file.")
